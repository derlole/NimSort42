	\section{Rahmenbedingungen}
	
	\begin{frame}{Rahmenbedingungen}
		\begin{itemize}
			\item Code für serielle Schnittstelle muss vorhanden sein
			\item Beleuchtung muss konstant und korrekt sein
			\item Kamera muss korrekt eingestellt sein
			\item Förderbandgeschwindigkeit konstant zwischen 0.01 m/s und 0.05 m/s
			\item Strom und Druckluftzufuhr
			\item Rechner mit ROS2 Humble muss konstant angeschlossen sein
		\end{itemize}
	\end{frame}
	
	\section{Performance}
	
	\begin{frame}{Performance}
		\begin{itemize}
			\item \textbf{Sortiergeschwindigkeit}: 0.01 m/s bis 0.03 m/s (optimal bei 0.015 m/s)

			\item \textbf{Objektklassifizierung}: 100\% (in Gesamt getestetem Zeitraum)
		\end{itemize}
