	\section{Designentscheidungen}										% Thema: Designentscheidungen	
		\begin{frame}{State Machines}
    \textbf{Designentscheidung:}

    State Machine für den gesamten Prozessablauf
    \vspace{0.5em}

    \textbf{Vorteile:}
    \begin{itemize}
        \item Prozessablauf ist sequentiell
        \item Klare Abgrenzung der Prozessschritte
        \item Einfaches Debugging
        \item Einfache Erweiterung
    \end{itemize}
    \vspace{0.5em}

    \textbf{Bewertung:}
    \begin{itemize}
        \item Komplexität gering
        \item Funktionsweise leicht verständlich
    \end{itemize}
\end{frame}

\begin{frame}{Sentinel}
    \textbf{Designentscheidung:}

    Sentinel senden anstatt keine Nachricht, falls kein Objekt erkannt wurde.
    Bei keiner Nachricht kontrolliertes Abbrechen des Prozesses.
    \vspace{0.5em}

    \textbf{Vorteile:}
    \begin{itemize}
        \item Prozessschritte können auf Sentinel reagieren
        \item Keine Abhängigkeit von der Reihenfolge der Nachrichten
        \item Einfaches Debugging
    \end{itemize}
    \vspace{0.5em}

    \textbf{Bewertung:}
    \begin{itemize}
        \item Funktioniert zuverlässig
        \item Simple Umsetzung
    \end{itemize}
\end{frame}

	\begin{frame}{PD-Regler vs. PDF-Regler}

		\textbf{Warum PD-Regler?}

		\begin{itemize}
			\item Einfache Parametrierung
			\item Stabiler Betrieb im Testaufbau
			\item Ausreichend gut
			\item Erweiterung auf PDF möglich
		\end{itemize}

		\vspace{0.8em}

		\textbf{Warum kein PDF-Regler?}

		\begin{itemize}
			\item Unsicherheiten mit dem Gefundenen Multiplikator 0.8
			\item Zusätzlicher Aufwand
			\item Nicht notwendig
		\end{itemize}

	\end{frame}
		
	\begin{frame}{Fail-Safe statt Fail-Operational}

	\textbf{Entscheidung}

	\vspace{0.5em}

	Verzicht auf Fail-Operational zugunsten eines
	definierten Safe-States mit anschließendem Neustart.

	\vspace{1em}
	\textbf{Warum?}
	\begin{itemize}
		\item Kritische Fehler können entlang der gesamten Verarbeitungskette entstehen
			(Vision $\rightarrow$ Prediction $\rightarrow$ Main $\rightarrow$ AxisController)
		\item Nach Fehler ist Neustart erforderlich
		\item Fail-Operational bringt wenig Vorteile aber höhere Komplexität % in dem speziellen Fall
		\item Ermöglicht automatisch Rückkehr in sichere Maschienenposition
	\end{itemize}

	\vfill
	\centering
	\Large
	Fehler erkannt $\rightarrow$ Safe-State $\rightarrow$ Neustart möglich

	\end{frame}

	\begin{frame}{Keine Aruco Marker – Homographie stattdessen}
	\textbf{Entscheidung:} Koordinatentransformation via Homographie statt Aruco Marker
		
	\vspace{1em}
	\begin{columns}[T]
	\begin{column}{0.48\textwidth}
	    \textit{Aruco Marker (verworfen)}
	    \begin{itemize}
			\item Pixel $\to$ Kamera $\to$ Welt via Marker-Pose
	        \item Positionsschätzung unzuverlässig
	        \item 3D-Transformation für 2D unnötig
	    \end{itemize}
	\end{column}
	\begin{column}{0.48\textwidth}
	    \textit{Homographie (gewählt)}
	    \begin{itemize}
	        \item Direkte Pixel $\to$ Welt Transformation
	        \item Robuster, kein Aruco Marker nötig
	        \item Für NimSort42 fölig ausreichend 
	    \end{itemize}
	\end{column}
	\end{columns}
	\end{frame}

	\begin{frame}{Warum Decision Tree?}
	\textbf{Merkmalsraum:} \texttt{hu\_0} vs.\ \texttt{hu\_3} zeigt klar getrennte Cluster

	\vspace{1em}
	\begin{itemize}
	    \item Klassen sind im Merkmalsraum kaum bzw. nicht überlappend
	    \item Trennungen verlaufen achsparallel
	    \item[$\Rightarrow$] Einfache Schwellwertentscheidungen der Form $hu\_0 < \theta$ genügen
	    \item[$\Rightarrow$] Decision Tree bildet genau diese Struktur ab
	\end{itemize}

	\vspace{1em}
	Komplexere Modelle (SVM, neuronales Netz) wären unnötiger Overhead, da sie deutlich rechenintensiver sind. 
	\end{frame}

	\begin{frame}{Homographie – Pixel zu Weltkoordinaten}
	\textbf{Ziel:} Pixelkoordinaten $\to$ reale Weltkoordinaten (mm) via $3{\times}3$-Matrix $H$

	\vspace{0.5em}
	\begin{columns}[T]
	\begin{column}{0.48\textwidth}
	    \textbf{Kalibrierung:}
	    \begin{itemize}
	        \item 5 Schachbrettquadrate, je 4 Eckpunkte
	        \item 20 Punktepaare (Pixel $\leftrightarrow$ mm) manuell vermessen
	        \item $H$ berechnet via \texttt{cv2.findHomography()}
	    \end{itemize}

	    \vspace{0.5em}
	    \textbf{Laufzeit:}
	    \begin{itemize}
	        \item Jede Pixelkoordinate wird mit $H$ transformiert
	        \item $H$ bleibt konstant solange Kamera unverändert
	    \end{itemize}

	    \vspace{0.5em}
	    \[
	        \mathbf{x}_{\mathrm{welt}} = H \cdot \mathbf{x}_{\mathrm{pixel}}
	    \]
	\end{column}
	\begin{column}{0.48\textwidth}
	    \includegraphics[width=\textwidth]{../../pictures/homography.jpg}
	\end{column}
	\end{columns}
	\end{frame}

	\begin{frame}{PD-Regler}

		\begin{columns}
		\begin{column}{0.55\textwidth}

		\textbf{Regelprinzip}
		\vspace{0.5em}

		\[
		e = x_{soll} - x_{ist}
		\]

		\[
		a = K_P \cdot e + K_D \cdot \dot{e}
		\]

		\[
		\dot{e} = \frac{(e - e_{old}) }{t_e - t_{e_{old}}}
		\]

		\vspace{0.5em}

		\begin{itemize}
			\item Ausgabe ist eine Beschleunigungsvorgabe
		\end{itemize}

		\end{column}
		\begin{column}{0.45\textwidth}
		\textbf{Ablauf}
		\vspace{0.5em}
		\begin{enumerate}
			\item Position empfangen
			\item Fehler berechnen
			\item Fehleränderung bestimmen
			\item Beschleunigung berechnen
			\item Stellgröße veröffentlichen
		\end{enumerate}
		\end{column}
		\end{columns}

		\vfill
		\centering
		\textbf{Ziel: Schnelle und stabile Annäherung an die Sollposition}

	\end{frame}

	\begin{frame}{Verarbeitungsgeschwindigkeit der Bildverarbeitung}	% Übernahme Louis
	\textbf{Problem:} Homographiematrix wurde für jedes Bild neu berechnet

	\vspace{1em}
	\begin{itemize}
	    \item Kameraposition bleibt im Betrieb konstant – $H$ ändert sich nie
	    \item[$\Rightarrow$] Berechnung einmalig in \texttt{\_\_init\_\_()} ausgelagert
	    \item[$\Rightarrow$] Deutliche Steigerung der Verarbeitungsgeschwindigkeit
	\end{itemize}
	\end{frame}

	\begin{frame}{Koordinatentransformation – Lessons Learned}
	\textbf{Bezug:} Entscheidung keine verwendung von Aruco Markern

	\vspace{1em}
	\begin{itemize}
	    \item Pixel $\to$ Kamera $\to$ Welt war in der Praxis zu fehleranfällig
	          (Bildverzerrung, aufwendige Kamerapose-Kalibrierung)
	    \item Homographie umgeht die Zwischentransformation: direkt Pixel $\to$ Welt
	    \item Kompromiss: Kamera muss vor jedem Start einmalig manuell ausgerichtet werden
	    \item[$\Rightarrow$] Deutlich geringerer Aufwand
	\end{itemize}
	\end{frame}

	\begin{frame}{Belichtung – Lessons Learned}
	\begin{itemize}
	    \item Pipeline funktionierte zunächst korrekt aber nach kurzer Zeit keine Erkennung mehr
	    \item Ursache: veränderte Belichtung (z.\,B. Sonneneinstrahlung, Raumbeleuchtung, etc.)
	    \item Konsistente Belichtung ist neben der Kameraausrichtung der wichtigste Faktor für zuverlässige Objekterkennung
	\end{itemize}
	\end{frame}
